async function minerarOrulo() { const url = document.getElementById('urlOrulo').value; const btn = document.getElementById('btnPublicar'); // LIMPA ANTES DE BUSCAR document.getElementById('editTitulo').value = "Minerando..."; const res = await fetch('/api/minerar-orulo', { method: 'POST', headers: {'Content-Type': 'application/json'}, body: JSON.stringify({ url }) }); const result = await res.json(); if(result.success) { alert("Imóvel extraído do Orulo com sucesso e publicado no site!"); } else { alert("Erro: " + result.error); } }
