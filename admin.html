async function publicarImovel() { const imovel = { titulo: document.getElementById('editTitulo').value, preco: document.getElementById('editPreco').value, foto: document.getElementById('editFoto').value, localizacao: document.getElementById('editLoc').value, area: "A definir", dormitorios: "A definir" }; const res = await fetch('/api/salvar-imovel', { method: 'POST', headers: {'Content-Type': 'application/json'}, body: JSON.stringify(imovel) }); if((await res.json()).success) { alert("Imóvel publicado com sucesso no site!"); } }
