// Função para disparar a mineração real async function processarOrulo() { const url = document.getElementById('oruloUrl').value; if (!url) return alert('Por favor, cole a URL do Órulo!'); // Mostra que está processando const btn = document.querySelector('.btn-submit'); const originalText = btn.innerText; btn.innerText = "⏳ MINERANDO DADOS... AGUARDE"; btn.disabled = true; try { const response = await fetch('/api/importar-orulo', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ url: url }) }); const data = await response.json(); if (data.success) { alert('✅ SUCESSO!\n\nImóvel: ' + data.imovel.titulo + '\nStatus: Publicado e postado nas Redes Sociais.'); window.location.href = "/"; // Volta para a home para ver o resultado } else { alert('❌ ERRO: ' + (data.error || 'Falha ao minerar o site.')); } } catch (error) { console.error(error); alert('❌ Erro de conexão com o servidor.'); } finally { btn.innerText = originalText; btn.disabled = false; } } // Vincula o clique do botão à função real document.querySelector('.btn-submit').onclick = processarOrulo; function fazerLogin() { const u = document.getElementById('user').value; const p = document.getElementById('pass').value; if (u === 'admin' && p === '1234') { document.getElementById('loginOverlay').style.display = 'none'; } else { alert('Incorreto! Use: admin / 1234'); } }
